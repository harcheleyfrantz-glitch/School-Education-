{$CLEO .csi}
// armour_steps: start it from the CLEO menu to add armour in ten steps of 10, using a counting
// loop and a subroutine. Original sample for Rodwave (MIT licence).
0000: nop
03A4: script_name 'ARMOUR'

if
    0256: is_player_playing $PLAYER_CHAR
then
    01F5: get_player_char $PLAYER_CHAR store_to 0@
    for 1@ = 1 to 10
        gosub @ADD_STEP
        wait 200
    end
end
004E: terminate_this_script

:ADD_STEP
035F: add_armour_to_char 0@ amount 10
return
