local buf, n = {}, 0
local function w(fmt, ...)
    n += 1
    if select("#", ...) > 0 then
        local okf, s = pcall(string.format, fmt, ...)
        buf[n] = okf and s or (tostring(fmt) .. "  <<fmt failed>>")
    else
        buf[n] = tostring(fmt)
    end
end

local function safe(fn, ...)
    local okc, res = pcall(fn, ...)
    if okc then return res end
    return nil
end

-- section runner: isolates failures so one bad section can't kill the dump ----
local ERRORS = {}
local function section(name, fn)
    local okr, err = xpcall(fn, function(e)
        return tostring(e) .. "\n" .. safe(function() return debug.traceback("", 2) end or ""
    end)
    if not okr then
        ERRORS[#ERRORS + 1] = name .. " -> " .. tostring(err)
        w("[[ERROR in %s]] %s", name, tostring(err))
    end
end

local function clip(s, len)
    len = len or MAX_ARG_LEN
    s = tostring(s)
    return (#s > len) and (s:sub(1, len) .. "...<cut>") or s
end

local function short(v, depth)
    depth = depth or 0
    if type(v) == "string" then return clip('"' .. v .. '"') end
    if type(v) == "function" or type(v) == "thread" then return "<" .. type(v) .. ">" end
    if type(v) == "table" then
        if depth > 3 then return "<table>" end
        local parts, count = {}, 0
        for k, val in pairs(v) do
            count += 1
            if count > 14 then parts[#parts + 1] = "..." break end
            parts[#parts + 1] = tostring(k) .. "=" .. short(val, depth + 1)
        end
        return "{ " .. table.concat(parts, ", ") .. " }"
    end
    return clip(v)
end

local GENERIC_PROPS = { "Value","Enabled","Locked","Visible","Transparency","Title","Priority" }

local function propBlock(inst)
    local out = {}
    local list = CLASS_PROPS[inst.ClassName] or GENERIC_PROPS
    for _, p in ipairs(list) do
        local v = safe(function() return inst[p] end)
        if v ~= nil and type(v) ~= "function" and type(v) ~= "thread" then
            if type(v) == "number" then v = string.format("%.3f", v) end
            out[#out + 1] = p .. "=" .. clip(tostring(v), 120)
        end
    end
    local at = safe(function() return inst:GetAttributes() end)
    if type(at) == "table" then
        local keys = {}
        for k in pairs(at) do keys[#keys + 1] = k end
        table.sort(keys)
        for _, k in ipairs(keys) do out[#out + 1] = "attr." .. clip(tostring(k), 40) .. "=" .. short(at[k]) end
    end
    local tags = safe(function() return CollectionService:GetTags(inst) end)
    if type(tags) == "table" and #tags > 0 then
        out[#out + 1] = "tags=[" .. table.concat(tags, ",") .. "]"
    end
    return #out > 0 and ("  {" .. table.concat(out, ", ") .. "}") or ""
end

-- assets ------------------------------------------------------------------
local assets, assetSeen = {}, {}
local function noteAsset(id, owner, kind)
    if type(id) ~= "string" or id == "" or id == "rbxassetid://" then return end
    local bare = id:gsub("^rbxassetid://", "")
    if not bare:match("^%-?%d+$") then return end
    if assetSeen[bare] then return end
    assetSeen[bare] = true
    assets[#assets + 1] = { id = "rbxassetid://" .. bare, owner = owner, kind = kind, num = bare }
end

local function collectAssets(root)
    for _, inst in ipairs(root:GetDescendants()) do
        local cn = inst.ClassName
        if cn == "Sound" then
            noteAsset(safe(function() return inst.SoundId end), inst:GetFullName(), "Sound")
        elseif cn == "Texture" or cn == "Decal" or cn == "Trail" or cn == "Beam" or cn == "ParticleEmitter" then
            noteAsset(safe(function() return inst.Texture end), inst:GetFullName(), cn)
        elseif cn == "MeshPart" then
            noteAsset(safe(function() return inst.MeshId end), inst:GetFullName(), "MeshId")
            noteAsset(safe(function() return inst.TextureID end), inst:GetFullName(), "MeshTexture")
        elseif cn == "SpecialMesh" then
            noteAsset(safe(function() return inst.TextureId end), inst:GetFullName(), "SpecialMesh")
        elseif cn == "Animation" then
            noteAsset(safe(function() return inst.AnimationId end), inst:GetFullName(), "AnimationId")
        elseif cn == "ImageLabel" or cn == "ImageButton" then
            noteAsset(safe(function() return inst.Image end), inst:GetFullName(), "Image")
        end
    end
end

-- tree --------------------------------------------------------------------
local function dumpTree(root, label)
    w("=== %s ===", label)
    local function walk(inst, depth)
        if depth > MAX_DEPTH then return end
        local pad = string.rep("  ", depth)
        w("%s[%s] %s%s", pad, inst.ClassName, inst.Name, propBlock(inst))
        local kids = safe(function() return inst:GetChildren() end) or {}
        if #kids > MAX_CHILDREN then
            w("%s  ...%d more children (truncated)", pad, #kids - MAX_CHILDREN)
            for i = 1, MAX_CHILDREN do walk(kids[i], depth + 1) end
        else
            for _, c in ipairs(kids) do walk(c, depth + 1) end
        end
    end
    walk(root, 0)
    w("")
end

-- remote logger ------------------------------------------------------------
local logLines, logN = {}, 0
local flushRemoteLog = function() end

local function log(fmt, ...)
    logN += 1
    if select("#", ...) > 0 then
        local okf, s = pcall(string.format, fmt, ...)
        logLines[logN] = okf and s or tostring(fmt)
    else
        logLines[logN] = tostring(fmt)
    end
    if logN % 250 == 0 then flushRemoteLog() end
end

local REMOTE_METHODS = { "FireServer", "FireClient", "FireAllClients", "InvokeServer", "InvokeClient" }

-- SAVE --------------------------------------------------------------------
local savedPaths = {}

local function tryWrite(path, data)
    if type(writefile) ~= "function" then return false, "no writefile" end
    if safe(function() return exists and exists(path) end) then return true, "exists" end
    pcall(function() writefile(path, data) end)
    local back = safe(function() return readfile(path) end)
    if type(back) == "string" and #back > 0 then return true, "wrote " .. #back .. "b" end
    return false, "write returned nothing"
end

local function saveAll(body)
    local remoteBody = table.concat(logLines, "\n")

    local function attempt(path, data)
        local good, why = tryWrite(path, data)
        if good then savedPaths[#savedPaths + 1] = path end
        return good, why, path
    end

    -- 1. Delta workspace files (relative path, next to the script)
    safe(function() return makefolder and makefolder(FOLDER_NAME) end)
    attempt(FOLDER_NAME .. "/" .. OUT_NAME, body)
    if logN > 0 then attempt(FOLDER_NAME .. "/" .. LOG_NAME, remoteBody) end

    -- 2. shared storage mirrors
    if ALSO_COPY_DOWNLOAD then
        for _, base in ipairs({
            "/storage/emulated/0/Download/",
            "/sdcard/Download/",
            "/storage/emulated/0/",
        }) do
            safe(function() return makefolder and makefolder(base .. FOLDER_NAME) end)
            attempt(base .. FOLDER_NAME .. "/" .. OUT_NAME, body)
            if logN > 0 then attempt(base .. FOLDER_NAME .. "/" .. LOG_NAME, remoteBody) end
        end
    end

    -- 3. executor fallbacks
    if #savedPaths == 0 and type(savefile) == "function" then
        safe(function() savefile(OUT_NAME, body) end)
        safe(function() savefile(FOLDER_NAME .. "/" .. OUT_NAME, body) end)
        savedPaths[#savedPaths + 1] = "savefile -> " .. OUT_NAME
    end
    if #savedPaths == 0 and type(setclipboard) == "function" then
        safe(function() setclipboard(body) end)
        savedPaths[#savedPaths + 1] = "clipboard (writes blocked)"
    end
end

flushRemoteLog = function()
    if logN == 0 or type(writefile) ~= "function" then return end
    local body = table.concat(logLines, "\n")
    for _, p in ipairs({
        FOLDER_NAME .. "/" .. LOG_NAME,
        "/storage/emulated/0/Download/" .. FOLDER_NAME .. "/" .. LOG_NAME,
    }) do
        safe(function() return makefolder and makefolder(FOLDER_NAME) end)
        safe(function() writefile(p, body) end)
    end
    print("[delta] remote log flushed (" .. logN .. " entries)")
end

    guiSet("building file...", Color3.fromRGB(255, 220, 120))

    local body = table.concat(buf, "\n")
    body = body .. "\n\n=== REMOTE LOG (live FireServer / InvokeServer / InvokeClient) ===\n"
    body = body .. (logN > 0 and table.concat(logLines, "\n")
        or "(nothing fired yet - play the game, then rerun this script for a fresh dump)")
    body = body .. "\n\n[END] lines=" .. n .. " remoteCalls=" .. logN .. " bytes=" .. #body .. "\n"

    guiSet("saving...", Color3.fromRGB(255, 220,120))
    saveAll(body)

    if #savedPaths > 0 then
        print("[delta] OK - " .. #body .. " bytes, " .. n .. " lines, " .. logN .. " remote calls")
        for _, s in ipairs(savedPaths) do print("[delta] saved: " .. s) end
        guiSet("DUMP SAVED\n" .. #body .. " bytes\n" .. logN .. " remote calls\n\nDelta files: game dump/\n\n" .. OUT_NAME,
               Color3.fromRGB(90, 255, 160))
    else
        warn("[delta] NO WRITE FUNCTION AVAILABLE - nothing written")
        guiSet("WRITE BLOCKED\nno writefile/savefile\ncheck executor permissions",
               Color3.fromRGB(255, 90, 90))
    end
    return savedPaths
end

-- REMOTE HOOKER ------------------------------------------------------------
if LOG_REMOTES then
    task.spawn(function()
        local hooked = {}
        local function hookRemote(remote)
            if hooked[remote] then return end
            hooked[remote] = true
            for _, meth in ipairs(REMOTE_METHODS) do
                if safe(function() return type(remote[meth]) == "function" end) then
                    local got = safe(function()
                        local orig = remote[meth]
                        remote[meth] = function(...)
                            local args = {}
                            for i = 1, select("#", ...) do args[i] = short((select(i, ...))) end
                            log("[%s] %s :%s(%s)",
                                os.date and os.date("%H:%M:%S") or "?",
                                remote:GetFullName(), meth, table.concat(args, ", "))
                            return orig(...)
                        end
                        return true
                    end)
                    if got == false then print("[delta] could not hook " .. remote:GetFullName()) end
                end
            end
        end

        local function sweep()
            for _, inst in ipairs(game:GetDescendants()) do
                if inst.ClassName:find("Remote") then hookRemote(inst) end
            end
        end

        pcall(sweep)
        while true do
            task.wait(3)
            pcall(sweep)
            if logN > 0 then flushRemoteLog() end
        end
    end)
end
